# Cron entries para smartbc en el VPS
# Agregar estas líneas al crontab del VPS (sudo crontab -e)

# Variables de entorno
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
CRON_SECRET=tu-secret-aqui
APP_DIR=/path/a/smartbc

# Scraper de particulares - cada hora
0 * * * * $APP_DIR/scripts/cron-particulares.sh >> $APP_DIR/logs/cron-particulares.log 2>&1

# Verificación de teléfonos en particulares - cada 2 días a las 00:00 UTC
0 0 */2 * * $APP_DIR/scripts/verify-particulares-phones.sh missing 100 >> $APP_DIR/logs/cron-verify-phones.log 2>&1

# Notas:
# 1. Reemplaza /path/a/smartbc con la ruta real del proyecto en el VPS
# 2. Reemplaza tu-secret-aqui con el valor real de CRON_SECRET
# 3. Asegúrate de que la carpeta logs/ existe: mkdir -p /path/a/smartbc/logs
# 4. Los permisos del script deben ser 755: chmod +x scripts/*.sh
